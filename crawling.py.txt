import os
import time
import requests
from selenium import webdriver
from bs4 import BeautifulSoup

# --- 1. 기본 설정 ---
# 수집할 키워드 (예: 연예인 이름)
keyword = "차은우" 
# 목표 수집 장수
target_count = 500 
# 저장 경로 설정 (./data/raw/키워드)
save_folder = f"./data/raw/{keyword}" 

# 저장 폴더가 없으면 새로 생성
if not os.path.exists(save_folder):
    os.makedirs(save_folder)

# --- 2. 브라우저 실행 및 접속 ---
# 크롬 드라이버 실행
driver = webdriver.Chrome()
# 요소 로딩 대기 시간 설정 (5초)
driver.implicitly_wait(5)
# 네이버 이미지 검색 URL 접속
url = f"https://search.naver.com/search.naver?where=image&query={keyword}"
driver.get(url)
# 페이지 로드 대기
time.sleep(2)

# --- 3. 스크롤 (데이터 확보) ---
print(f"{keyword} 사진 수집을 위한 스크롤 시작")
# 지정된 횟수만큼 화면을 아래로 내려 이미지 더 불러오기
for i in range(30): 
    driver.execute_script("window.scrollTo(0, document.body.scrollHeight)")
    # 스크롤 후 이미지 로딩 시간 대기
    time.sleep(1.5) 

# --- 4. 이미지 URL 추출 ---
# 현재 페이지의 HTML 소스 가져오기
soup = BeautifulSoup(driver.page_source, "html.parser")
# 이미지 태그 선택 (네이버 이미지 검색의 특정 셀렉터 사용)
imgs = soup.select(".image_wrap img[src]")
# 실제 이미지 주소(src)만 리스트에 저장 (data 형식 제외)
img_urls = [img["src"] for img in imgs if not img["src"].startswith("data")]
# 중복된 URL 제거
img_urls = list(set(img_urls)) 
print(f"총 {len(img_urls)}개의 이미지 주소 확보")

# --- 5. 이미지 파일 저장 ---
# 봇 차단 방지를 위한 브라우저 정보 설정
headers = {"User-Agent": "Mozilla/5.0"}
success_count = 0

for i, url in enumerate(img_urls):
    # 목표 장수를 채우면 반복 중단
    if success_count >= target_count: 
        break
        
    # 저장될 파일명 형식 (예: 강아지_001.jpg)
    file_name = f"{keyword}_{success_count+1:03}.jpg" 
    path = os.path.join(save_folder, file_name)
    
    try:
        # 이미지 주소로 요청 전송
        r = requests.get(url, headers=headers, timeout=10)
        # 정상 응답(200)일 때만 파일로 저장
        if r.status_code == 200:
            with open(path, "wb") as f:
                f.write(r.content)
            success_count += 1
            # 50장마다 중간 진행 상황 출력
            if success_count % 50 == 0:
                print(f"{success_count}장 저장 완료")
    except:
        # 다운로드 에러 발생 시 건너뛰고 계속 진행
        continue

# 모든 작업 완료 후 브라우저 종료
driver.quit()
print(f"최종 {success_count}장 저장 완료")

import os

random.seed(42) 

# --- 1. 경로 및 설정 ---
source_base = r"D:\project4\data"          
split_base = r"D:\project4\data\split"

train_ratio = 0.8
val_ratio = 0.1
test_ratio = 0.1

print("[3단계] 8인 데이터 분할 시작")

for celeb in celebs:
    for emotion in emotions:
        src_path = os.path.join(source_base, celeb, emotion)
        
        if not os.path.exists(src_path):
            continue
            
        # 1. 파일 목록 추출 및 랜덤 섞기
        files = [f for f in os.listdir(src_path) if f.lower().endswith(('.jpg', '.jpeg', '.png'))]
        random.shuffle(files) # 시드(Seed)는 맨 위에서 고정했으니 여기선 섞기만 함
        
        # 2. 비율에 따른 파일 개수 계산
        total = len(files)
        if total == 0:
            continue
            
        num_train = int(total * train_ratio)
        num_val = int(total * val_ratio)
        
        # 3. 데이터 구간 나누기 (슬라이싱)
        train_files = files[:num_train]
        val_files = files[num_train : num_train + num_val]
        test_files = files[num_train + num_val :]
        
        # 4. 파일 복사 실행
        file_sets = [('train', train_files), ('val', val_files), ('test', test_files)]
        
        for set_name, set_files in file_sets:
            target_path = os.path.join(split_base, set_name, celeb, emotion)
            os.makedirs(target_path, exist_ok=True) 
            
            for f in set_files:
                src_file = os.path.join(src_path, f)
                dst_file = os.path.join(target_path, f)
                shutil.copy(src_file, dst_file) 
        
        print(f"{celeb} ({emotion}): 총 {total}장 (Train:{len(train_files)} / Val:{len(val_files)} / Test:{len(test_files)}) 분할 완료")

print("-" * 50)
print("데이터 분할 작업 끝")

import os
from PIL import Image
# 설치한 인공지능 (MTCNN) 가져오기
from facenet_pytorch import MTCNN 

# --- 1. 경로 및 도구 설정 ---
# 3단계에서 나눈 사진들이 있는 곳
split_base = r"D:\project4\data\split"          
# 얼굴만 오려낸 결과물을 저장할 새 폴더
cropped_base = r"D:\project4\data\cropped"      

# 얼굴 탐지기(MTCNN) 설정
# keep_all=False: 사진 속에서 가장 큰 얼굴 하나만 찾음
mtcnn = MTCNN(keep_all=False, select_largest=True, device='cpu')



PADDING_RATIO = 0.20  # 여백을 20%로 살짝 감소
MIN_AREA_RATIO = 0.1  # 얼굴 면적 비율을 10%로 대폭 완화

print("MTCNN 얼굴 추출 및 전처리 시작...")

# --- 2. 폴더를 돌며 사진 하나씩 처리 ---
for split_name in ['train', 'val', 'test']:
    split_dir = os.path.join(split_base, split_name)
    if not os.path.exists(split_dir): continue

    # 연예인 이름 폴더 확인
    for celeb in os.listdir(split_dir):
        # smile, neutral 폴더 확인
        for emotion in ['smile', 'neutral']:
            src_path = os.path.join(split_dir, celeb, emotion)
            target_path = os.path.join(cropped_base, split_name, celeb, emotion)
            
            if not os.path.exists(src_path): continue
            os.makedirs(target_path, exist_ok=True) # 저장할 폴더 생성

            success_cnt = 0
            fail_cnt = 0
            files = [f for f in os.listdir(src_path) if f.lower().endswith(('.jpg', '.jpeg', '.png'))]

            for img_name in files:
                img_path = os.path.join(src_path, img_name)
                
                try:
                    # 사진 열기
                    img = Image.open(img_path).convert('RGB')
                    img_width, img_height = img.size
                    img_area = img_width * img_height

                    # --- 3. 얼굴 위치 찾기 ---
                    boxes, _ = mtcnn.detect(img)

                    if boxes is None:
                        fail_cnt += 1 # 얼굴을 못 찾으면 제외
                        continue
                        
                    box = boxes[0] # 가장 큰 얼굴 좌표 [왼쪽, 위, 오른쪽, 아래]
                    
                    # --- 4. [기획서 조건 1] 얼굴 면적 비율 검사 ---
                    face_width = box[2] - box[0]
                    face_height = box[3] - box[1]
                    face_area = face_width * face_height
                    
                    # 얼굴이 사진 전체의 30%보다 작으면 '배경 편향' 방지를 위해 버림
                    if (face_area / img_area) < MIN_AREA_RATIO:
                        fail_cnt += 1
                        continue

                    # --- 5.여백(Padding) 계산 ---
                    pad_w = face_width * PADDING_RATIO
                    pad_h = face_height * PADDING_RATIO

                    # 사진 범위를 벗어나지 않도록 상하좌우 여백 추가
                    left = max(0, box[0] - pad_w)
                    top = max(0, box[1] - pad_h)
                    right = min(img_width, box[2] + pad_w)
                    bottom = min(img_height, box[3] + pad_h)

                    # --- 6. 자르기(Crop) 및 저장 ---
                    cropped_img = img.crop((left, top, right, bottom))
                    # 모델 학습 규격인 224x224 크기로 통일해서 저장
                    cropped_img = cropped_img.resize((224, 224), Image.LANCZOS)
                    cropped_img.save(os.path.join(target_path, img_name))
                    success_cnt += 1

                except Exception:
                    fail_cnt += 1

            # [기획서 조건 3] 클래스별 개수 통계 출력
            print(f"[{split_name}] {celeb} ({emotion}) - 성공: {success_cnt}장 / 제외됨: {fail_cnt}장")

print("-" * 50)
print("모든 얼굴 추출 작업이 완료되었습니다! 'data/cropped' 폴더를 확인하세요.")

import os
from PIL import Image
import torch
from torch.utils.data import Dataset, DataLoader
from torchvision import transforms

# --- 1. Train / Val 전용 Transform(데이터 증강 및 전처리) ---
def get_transforms():
    data_transforms = {
        'train': transforms.Compose([
            transforms.RandomHorizontalFlip(),                     
            transforms.RandomRotation(10),                         
            transforms.ColorJitter(brightness=0.2, contrast=0.2),  
            transforms.ToTensor(),                                 
            transforms.Normalize([0.485, 0.456, 0.406], 
                                 [0.229, 0.224, 0.225])            
        ]),
        'val': transforms.Compose([
            transforms.Resize((224, 224)),                         
            transforms.ToTensor(),
            transforms.Normalize([0.485, 0.456, 0.406], 
                                 [0.229, 0.224, 0.225])
        ]),
    }
    return data_transforms

# --- 2. 멀티태스크 전용 커스텀 데이터셋 (여기서 3개를 반환합니다!) ---
class MultiTaskDataset(Dataset):
    def __init__(self, root_dir, transform=None):
        self.root_dir = root_dir
        self.transform = transform
        self.image_paths = []
        self.celeb_labels = []
        self.smile_labels = []

        # 1) 연예인 폴더(Class) 추출
        self.celebs = sorted(os.listdir(root_dir))
        self.celeb_to_idx = {c: i for i, c in enumerate(self.celebs)}
        
        # 2) 표정 폴더(Class) 설정 (기획안 기준 2개)
        self.emotions = ['neutral', 'smile']
        self.emotion_to_idx = {e: i for i, e in enumerate(self.emotions)}

        # 3) 폴더를 돌면서 "사진 경로, 연예인 번호, 표정 번호" 3가지를 싹 다 수집
        for celeb in self.celebs:
            celeb_path = os.path.join(root_dir, celeb)
            if not os.path.isdir(celeb_path): continue

            for emotion in self.emotions:
                emotion_path = os.path.join(celeb_path, emotion)
                if not os.path.isdir(emotion_path): continue

                for img_name in os.listdir(emotion_path):
                    if img_name.lower().endswith(('.png', '.jpg', '.jpeg')):
                        self.image_paths.append(os.path.join(emotion_path, img_name))
                        self.celeb_labels.append(self.celeb_to_idx[celeb])
                        self.smile_labels.append(self.emotion_to_idx[emotion])

    def __len__(self):
        return len(self.image_paths)

    def __getitem__(self, idx):
        # 사진 꺼내기
        img_path = self.image_paths[idx]
        image = Image.open(img_path).convert('RGB')

        if self.transform:
            image = self.transform(image)

        # 정답지 2개 꺼내기
        celeb_label = self.celeb_labels[idx]
        smile_label = self.smile_labels[idx]

        # 
        return image, celeb_label, smile_label

# --- 3. DataLoader 생성 함수 ---
def get_dataloaders(data_dir, batch_size=32):
    transforms_dict = get_transforms()

    train_dir = os.path.join(data_dir, 'train')
    val_dir = os.path.join(data_dir, 'val')

    train_dataset = MultiTaskDataset(train_dir, transform=transforms_dict['train'])
    val_dataset = MultiTaskDataset(val_dir, transform=transforms_dict['val'])

    dataloaders = {
        'train': DataLoader(train_dataset, batch_size=batch_size, shuffle=True),
        'val': DataLoader(val_dataset, batch_size=batch_size, shuffle=False)
    }

    class_names = train_dataset.celebs

    print(f"\n[Dataset] 멀티태스크 데이터 로드 완료! (연예인: {len(class_names)}명, 표정: 2개)")
    print(f" - Train 데이터: {len(train_dataset)}장")
    print(f" - Val 데이터: {len(val_dataset)}장\n")

    return dataloaders, class_names

if __name__ == '__main__':
    print("dataset.py 세팅 완료.")

import torch
import torch.nn as nn
from torchvision import models

# 연예인 이름(8명)과 웃음 여부(2개)를 동시에 맞히는 멀티태스크 모델
class MultiTaskResNet18(nn.Module):
    def __init__(self, num_celebs=8):
        super(MultiTaskResNet18, self).__init__()
        
        # 1. 전이 학습용 ResNet18 로드 (미리 학습된 가중치 사용)
        # weights='IMAGENET1K_V1'은 이미 수많은 사물을 공부한 상태임을 의미함
        self.resnet = models.resnet18(weights='IMAGENET1K_V1')
        
        # ResNet18의 마지막 층(fc) 직전의 출력값 크기를 가져옴 (보통 512)
        in_features = self.resnet.fc.in_features
        
        # 2. 원래 있던 마지막 출력층 제거 (Identity로 대체)
        # 뇌(특징 추출기)만 쓰고, 입(분류기)은 우리가 새로 달기 위함
        self.resnet.fc = nn.Identity()
        
        # 3. 과적합 방지를 위한 드롭아웃(Dropout) 추가
        # 학습 시 50%의 뉴런을 무작위로 꺼서 특정 특징에만 집착하지 않게 함
        self.dropout = nn.Dropout(p=0.5)
        
        # 4. 멀티태스크용 두 개의 머리 (Fully Connected Layers)
        # Head 1: 연예인 8명 분류
        self.fc_celeb = nn.Linear(in_features, num_celebs)
        # Head 2: 웃음/무표정 2개 클래스 분류
        self.fc_smile = nn.Linear(in_features, 2)

    def forward(self, x):
        # 1. ResNet18의 뇌를 통과하며 이미지 특징 추출
        x = self.resnet(x)
        
        # 2. 드롭아웃 적용 (훈련 중에만 동작)
        x = self.dropout(x)
        
        # 3. 각각의 입으로 정보를 전달하여 두 가지 결과 도출
        out_celeb = self.fc_celeb(x) # 연예인 누구인지 (아웃 셀럽)
        out_smile = self.fc_smile(x) # 웃는지 안 웃는지 (아웃 스마일)
        
        return out_celeb, out_smile

import torch
import torch.nn as nn
import torch.optim as optim
import os
import matplotlib.pyplot as plt
from tqdm import tqdm 
from dataset import get_dataloaders   
from model import MultiTaskResNet18      

# --- 1. 설정 및 데이터 로드 ---
device = torch.device("cuda" if torch.cuda.is_available() else "cpu")
root_dir = r"D:\project4\data\cropped" 
batch_size = 32
epochs = 200  
alpha, beta = 1.0, 0.5  

# 만든 dataset.py 규칙에 맞게 데이터를 꺼내옵니다.
dataloaders, class_names = get_dataloaders(root_dir, batch_size)
train_loader = dataloaders['train']
val_loader = dataloaders['val']
num_celebs = len(class_names)

# (모델) 불러오기
model = MultiTaskResNet18(num_celebs=num_celebs).to(device)

criterion = nn.CrossEntropyLoss()
optimizer = optim.Adam(model.parameters(), lr=0.001)

# 점수가 안 오르면 학습 속도를 줄여주는 스케줄러
scheduler = optim.lr_scheduler.ReduceLROnPlateau(optimizer, mode='max', factor=0.5, patience=3, verbose=True)

# 학습 기록 저장용 
history = {'train_loss': [], 'val_loss': [], 'celeb_acc': [], 'smile_acc': []}

print(f"실전 학습 시작 (장치: {device})")

# --- 2. 학습 루프 (이 아래는 기존 코드 그대로 유지) ---

# --- 2. 학습 루프(Loop, 루프) ---
best_acc = 0.0
patience = 7  # 조기 종료(Early Stopping, 얼리 스토핑) 인내심
counter = 0

for epoch in range(epochs):
    # [Train 단계]
    model.train()
    train_loss = 0.0
    
    # 케라스(Keras, 케라스)처럼 게이지(Gauge, 게이지)가 차오름
    pbar = tqdm(train_loader, desc=f"Epoch {epoch+1:03d}/{epochs}")
    for images, celeb_labels, smile_labels in pbar:
        images = images.to(device)
        celeb_labels = celeb_labels.to(device)
        smile_labels = smile_labels.to(device)
        
        optimizer.zero_grad()
        out_celeb, out_smile = model(images)
        
        # 가중치 손실(Weighted Loss, 웨이티드 로스) 계산
        loss = (alpha * criterion(out_celeb, celeb_labels)) + (beta * criterion(out_smile, smile_labels))
        loss.backward()
        optimizer.step()
        
        train_loss += loss.item()
        pbar.set_postfix(loss=f"{loss.item():.4f}")

    avg_train_loss = train_loss / len(train_loader)

    # [Validation 단계]
    model.eval()
    val_loss = 0.0
    val_celeb_correct = 0
    val_smile_correct = 0
    total = 0
    
    with torch.no_grad():
        for images, celeb_labels, smile_labels in val_loader:
            images = images.to(device)
            celeb_labels = celeb_labels.to(device)
            smile_labels = smile_labels.to(device)
            
            out_celeb, out_smile = model(images)
            
            # 검증 손실(Validation Loss, 밸리데이션 로스) 추적
            v_loss = (alpha * criterion(out_celeb, celeb_labels)) + (beta * criterion(out_smile, smile_labels))
            val_loss += v_loss.item()
            
            _, pred_celeb = torch.max(out_celeb, 1)
            _, pred_smile = torch.max(out_smile, 1)
            
            val_celeb_correct += (pred_celeb == celeb_labels).sum().item()
            val_smile_correct += (pred_smile == smile_labels).sum().item()
            total += celeb_labels.size(0)

    # 결과 기록 및 계산
    avg_val_loss = val_loss / len(val_loader)
    c_acc = val_celeb_correct / total
    s_acc = val_smile_correct / total
    current_total_acc = c_acc + s_acc 
    
    history['train_loss'].append(avg_train_loss)
    history['val_loss'].append(avg_val_loss)
    history['celeb_acc'].append(c_acc)
    history['smile_acc'].append(s_acc)

    print(f" 결과 -> Train Loss: {avg_train_loss:.4f} | Val Loss: {avg_val_loss:.4f} | 이름: {c_acc:.4f} | 표정: {s_acc:.4f}")

    # 스케줄러(Scheduler, 스케줄러) 적용 (평균 점수를 기준으로 평가)
    scheduler.step(current_total_acc / 2)

    # 베스트 모델 저장 및 조기 종료
    if current_total_acc > best_acc:
        if s_acc >= 0.70:
            best_acc = current_total_acc
            torch.save(model.state_dict(), "best_model.pth")
            print(f"[안전장치 통과] 표정 정확도 {s_acc:.4f} (>=0.70) 만족! 최고 성능 갱신 및 저장 완료")
            counter = 0
        else:
            counter += 1
            print(f"전체 점수는 올랐으나, 표정 정확도({s_acc:.4f})가 0.70 미만이라 저장을 보류합니다.")
            if counter >= patience:
                print(f"\n{patience}번 연속 유효한 성능 향상이 없어 조기 종료(Early Stopping) 합니다.")
                break
    else:
        counter += 1
        if counter >= patience:
            print(f"\n{patience}번 연속 성능 향상이 없어 조기 종료(Early Stopping) 합니다.")
            break
